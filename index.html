window.addEventListener('keydown', e => keys[e.code] = true); window.addEventListener('keyup', e => keys[e.code] = false); function gameLoop() { ctx.clearRect(0, 0, canvas.width, canvas.height); const isAllLoaded = sprites.south.loaded && sprites.north.loaded && sprites.east.loaded; if (!isAllLoaded) { ctx.fillStyle = '#fff'; ctx.font = '16px sans-serif'; ctx.textAlign = 'center'; ctx.fillText('스프라이트 이미지를 불러오는 중입니다...', canvas.width / 2, canvas.height / 2); } else { isMoving = false; // 상하좌우 이동 조작 if (keys['ArrowDown']) { charY += 3; currentDirection = 'south'; isMoving = true; } else if (keys['ArrowUp']) { charY -= 3; currentDirection = 'north'; isMoving = true; } else if (keys['ArrowRight']) { charX += 3; currentDirection = 'east'; isMoving = true; } else if (keys['ArrowLeft']) { charX -= 3; currentDirection = 'west'; isMoving = true; } // 왼쪽 이동 시 east_walk.png를 반전해서 사용 const activeKey = (currentDirection === 'west') ? 'east' : currentDirection; const active = sprites[activeKey]; if (isMoving) { frameTimer++; if (frameTimer % 8 === 0) { frameIndex = (frameIndex + 1) % active.walkFrames.length; } } else { frameIndex = 0; } const currentFrame = active.walkFrames[frameIndex]; const sx = (currentFrame % active.cols) * active.frameW; const sy = Math.floor(currentFrame / active.cols) * active.frameH; const offsetY = isMoving ? active.offsets[frameIndex] : 0; const renderWidth = (active.frameW / active.frameH) * RENDER_HEIGHT; ctx.save(); if (currentDirection === 'west') { // 왼쪽 이동 시 X축 반전 렌더링 ctx.translate(charX + renderWidth, charY + offsetY); ctx.scale(-1, 1); ctx.drawImage( active.img, sx, sy, active.frameW, active.frameH, 0, 0, renderWidth, RENDER_HEIGHT ); } else { // 일반 렌더링 (south, north, east) ctx.drawImage( active.img, sx, sy, active.frameW, active.frameH, charX, charY + offsetY, renderWidth, RENDER_HEIGHT ); } ctx.restore(); } requestAnimationFrame(gameLoop); } requestAnimationFrame(gameLoop);
